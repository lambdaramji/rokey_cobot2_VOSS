### 1-1. 계산 원리 (box_tracker)
J6 회전 보정을 따로 하지 않는다 — 카메라 회전은 촬영 시각 pose 에 이미 들어 있으니, **박스 중심과 같은 변환으로 송장 긴 변 양 끝 두 점을 베이스로 옮겨 atan2** 하면 베이스 기준 각이 나온다.

```python
# box_detect.py — minAreaRect 꼭짓점에서 긴 변을 고른다 (OpenCV 각도 규칙에 기대지 않음)
def half_long_axis(rot) -> tuple[float, float]:
    c = cv2.boxPoints(rot)
    e1, e2 = c[1] - c[0], c[2] - c[1]
    e = e1 if np.hypot(*e1) >= np.hypot(*e2) else e2
    return float(e[0] / 2), float(e[1] / 2)

px = [[u, v]]
if half_axis is not None:
    hx, hy = half_axis
    px += [[u - hx, v - hy], [u + hx, v + hy]]
...
m.yaw_base, m.yaw_valid = tl.choose_yaw(src, valid, he_ends, homo_ends)

# tracker_logic.py
def fold_yaw(yaw):                       # rad → (−π/2, π/2]. 파지 면이 180° 대칭
    y = math.fmod(yaw, math.pi)
    if y <= -math.pi / 2: y += math.pi
    elif y > math.pi / 2: y -= math.pi
    return y

def choose_yaw(src, position_valid, he_ends, homo_ends):
    ends = {SRC_HAND_EYE: he_ends, SRC_HOMOGRAPHY: homo_ends}.get(src)   # position 과 같은 출처만
    if not position_valid or ends is None:
        return 0.0, False
    yaw = yaw_from_ends(*ends)            # 두 점이 1 mm 미만이면 None
    return (0.0, False) if yaw is None else (yaw, True)
```

### 3-2. 목표 각 (순수 함수 + pytest) — 2θ 중앙값
```python
# 제안: voss_servo/yaw.py
import math
import numpy as np

def fold(a: float) -> float:
    """(−π/2, π/2] — tracker_logic.fold_yaw 와 같은 규칙."""
    y = math.fmod(a, math.pi)
    return y + math.pi if y <= -math.pi / 2 else (y - math.pi if y > math.pi / 2 else y)

def yaw_target(yaws: list[float], min_n: int = 5, min_r: float = 0.9) -> float | None:
    """PREPARE 동안 모은 yaw_base(rad, yaw_valid 만) → 목표 박스 각. 부족·흩어짐이면 None(돌리지 않음).
    ±90° 경계에서 +89°·−89° 가 섞여도 2θ 로 두 배 하면 같은 쪽에 모인다."""
    if len(yaws) < min_n:
        return None
    a = 2.0 * np.asarray(yaws)
    c, s = np.cos(a), np.sin(a)
    if math.hypot(c.mean(), s.mean()) < min_r:     # 평균 합벡터 길이 — 흩어지면 믿지 않음
        return None
    return fold(0.5 * math.atan2(float(np.median(s)), float(np.median(c))))

def j6_delta(box_yaw: float, belt_dir_xy, deadband_deg: float = 4.0, max_deg: float = 90.0) -> float:
    """지금 그리퍼(닫힘축이 벨트 가로 = 박스 긴 변이 벨트 방향일 때 맞음) 기준으로 돌릴 베이스 Z 회전량 rad.
    데드밴드 안이면 0. 180° 대칭이라 |값| ≤ 90°."""
    belt_yaw = math.atan2(belt_dir_xy[1], belt_dir_xy[0])
    rel = fold(box_yaw - belt_yaw)
    if abs(math.degrees(rel)) < deadband_deg or abs(math.degrees(rel)) > max_deg:
        return 0.0
    return rel
```
```python
# 제안: test/test_yaw.py
import math
from voss_servo.yaw import j6_delta, yaw_target

D = math.radians

def test_median_survives_the_fold() -> None:      # +89°·−89° 섞임 → 90° 근처, 0° 가 아님
    t = yaw_target([D(89), D(-89), D(88), D(-88), D(89)])
    assert abs(abs(math.degrees(t)) - 89) < 1.5

def test_too_few_or_scattered_gives_none() -> None:
    assert yaw_target([D(10)] * 4) is None
    assert yaw_target([D(0), D(45), D(-45), D(90), D(30)]) is None

def test_deadband_and_belt_relative() -> None:
    belt = (1.0, 0.0)                               # 벨트 +X
    assert j6_delta(D(3), belt) == 0.0              # 데드밴드
    assert math.isclose(j6_delta(D(30), belt), D(30))
    assert math.isclose(j6_delta(D(-30), (0.0, 1.0)), D(60), abs_tol=1e-9)  # 벨트 +Y 기준 상대각
```

### 3-3. 각속도 명령 — `servo_cmd.twist.angular.z` (베이스 Z, rad/s)
`/voss/robot/servo_cmd` 는 이미 TwistStamped(베이스 기준 선속도·**각속도**)라 **인터페이스 변경은 없다.** 지금 `control.command()` 는 선속도만 만든다(`Command.vel`). 같은 틱에서 각속도를 따로 만든다:
```python
def tool_yaw(q_xyzw) -> float:
    """/voss/robot/pose 쿼터니언 → 공구 x 축의 베이스 XY 각 (공구가 아래를 볼 때)."""
    x, y, z, w = q_xyzw
    r00 = 1 - 2 * (y * y + z * z)
    r10 = 2 * (x * y + z * w)
    return math.atan2(r10, r00)

def wz_cmd(psi_target, psi_now, prev_wz, kp, w_max, a_max, dt) -> float:
    e = math.atan2(math.sin(psi_target - psi_now), math.cos(psi_target - psi_now))   # −π..π 감기
    w = max(-w_max, min(w_max, kp * e))
    dw = max(-a_max * dt, min(a_max * dt, w - prev_wz))                               # 각가속 상한
    return prev_wz + dw
```
- **언제** (TR-SERVO-05): goal 받고 PREPARE 동안 `yaw_target` 을 모아 **TRACK 들어갈 때 고정**(`psi_home` = 그때 공구 yaw, `psi_target = psi_home + j6_delta`). **TRACK 초반에 돌리고, `aligned` 조건에 `|e| < 허용` 추가**(정렬 전에 회전 끝). DESCEND·GRASP·접근 높이에서는 `wz = 0`. **LIFT 에서 `psi_home` 으로 되돌린다**(트레이에 정렬된 자세로 놓이게).
- 베이스 Z 각속도로 내면 J6 부호(공구가 아래라 J6 + 가 베이스 Z − 일 수 있음)는 gateway·컨트롤러가 푼다 — **이 가정을 에뮬레이터에서 확인**(3-5).
- 박스 각은 추종 중에 다시 재지 않는다(카메라가 돌며 바뀌는 값은 같은 박스라 같아야 정상 — 로그로만 비교).
